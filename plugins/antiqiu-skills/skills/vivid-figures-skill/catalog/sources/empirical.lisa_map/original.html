   8 import numpy as np
   9 import matplotlib.pyplot as plt
  10 from _utils.plot_utils import setup_style, save_fig, PALETTE, COLORS, _lighten
  11 setup_style()
  12 
  13 import geopandas as gpd
  14 from libpysal.weights import Queen
  15 from esda.moran import Moran_Local
  16 
  17 gdf = gpd.read_file('_utils/china_provinces.geojson')
  18 
  19 # 模拟有空间自相关的数据
  20 np.random.seed(42)
  21 gdf['centroid_x'] = gdf.geometry.centroid.x
  22 gdf['value'] = 0.3 + 0.005 * (gdf['centroid_x'] - 80) + np.random.randn(len(gdf)) * 0.08
  23 gdf['value'] = gdf['value'].clip(0.1, 0.95)
  24 
  25 # 构建空间权重 + 计算 LISA
  26 w = Queen.from_dataframe(gdf, use_index=False)
  27 w.transform = 'r'
  28 lisa = Moran_Local(gdf['value'].values, w, permutations=999)
  29 
  30 # LISA 聚类分类
  31 # lisa.q: 1=HH, 2=LH, 3=LL, 4=HL
  32 # lisa.p_sim: p 值
  33 sig_level = 0.05
  34 gdf['lisa_cluster'] = 'NS'
  35 for i in range(len(gdf)):
  36     if lisa.p_sim[i] <= sig_level:
  37         q = lisa.q[i]
  38         if q == 1: gdf.loc[gdf.index[i], 'lisa_cluster'] = 'HH'
  39         elif q == 2: gdf.loc[gdf.index[i], 'lisa_cluster'] = 'LH'
  40         elif q == 3: gdf.loc[gdf.index[i], 'lisa_cluster'] = 'LL'
  41         elif q == 4: gdf.loc[gdf.index[i], 'lisa_cluster'] = 'HL'
  42 
  43 cluster_colors = {
  44     'HH': '#E25B5B',  # 热点（红色）
  45     'LL': '#5B8DB8',  # 冷点（蓝色）
  46     'LH': '#A8C4D8',  # 低高异常（浅蓝）
  47     'HL': '#E0A0A0',  # 高低异常（浅红）
  48     'NS': '#F0F0F0',  # 不显著（灰色）
  49 }
  50 
  51 from collections import Counter
  52 counts = Counter(gdf['lisa_cluster'])
  53 
  54 # === 画图 ===
  55 fig, ax = plt.subplots(figsize=(10, 8))
  56 
  57 for cluster_type in ['NS', 'HL', 'LH', 'LL', 'HH']:  # NS 先画，显著的后画
  58     subset = gdf[gdf['lisa_cluster'] == cluster_type]
  59     if len(subset) > 0:
  60         subset.plot(color=cluster_colors[cluster_type], edgecolor='white',
  61                     linewidth=0.5, ax=ax, zorder=2 if cluster_type != 'NS' else 1)
  62

  77     c = row.geometry.centroid
