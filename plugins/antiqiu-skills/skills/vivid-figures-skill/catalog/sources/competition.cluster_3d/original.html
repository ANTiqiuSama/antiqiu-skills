1 import numpy as np
   2 import matplotlib.pyplot as plt
   3 from mpl_toolkits.mplot3d import Axes3D
   4 from _utils.plot_utils import setup_style, save_fig, PALETTE, COLORS
   5 setup_style()
   6 np.random.seed(42)
   7 
   8 # 生成 3D 聚类数据
   9 centers = np.array([[2,3,4], [6,2,7], [4,7,2], [8,6,5]])
  10 n_per = 60
  11 X = np.vstack([c + np.random.randn(n_per, 3)*0.8 for c in centers])
  12 labels = np.repeat(range(4), n_per)
  13 
  14 fig = plt.figure(figsize=(8, 6))
  15 ax = fig.add_subplot(111, projection='3d')
  16 
  17 for k in range(4):
  18     mask = labels == k
  19     ax.scatter(X[mask,0], X[mask,1], X[mask,2], s=20, alpha=0.5,
  20               color=PALETTE[k], edgecolor='white', linewidth=0.3, label=f'簇 {k+1}')
  21     ax.scatter(*centers[k], s=150, color=PALETTE[k], marker='X',
  22               edgecolor='white', linewidth=2, zorder=5)
  23 
  24 ax.set_xlabel('维度 1', fontsize=11, labelpad=6)
  25 ax.set_ylabel('维度 2', fontsize=11, labelpad=6)
  26 ax.set_zlabel('维度 3', fontsize=11, labelpad=8)
  27 ax.view_init(elev=20, azim=45)
  28 ax.legend(loc='upper left', fontsize=10, framealpha=0.9)
  29 fig.tight_layout()
  30 save_fig(fig, 'figures/fig_3d_cluster.pdf')
