1 import numpy as np, matplotlib.pyplot as plt
   2 from _utils.plot_utils import setup_style, save_fig, PALETTE, COLORS, _lighten
   3 setup_style()
   4 
   5 np.random.seed(42)
   6 years = np.arange(2010, 2024)

  11 
  12 n_layers = 12
  13 for k in range(n_layers, 0, -1):
  14     frac = k / n_layers
  15     ax1.fill_between(years, gdp.min()-0.5, gdp-(gdp-gdp.min()+0.5)*(1-frac),
  16                      alpha=0.02, color=PALETTE[0], linewidth=0)
  17 ax1.fill_between(years, gdp.min()-0.5, gdp, alpha=0.08, color=PALETTE[0], linewidth=0)
  18 
  19 for k in range(n_layers, 0, -1):
  20     frac = k / n_layers
  21     ax2.fill_between(years, digital.min()-0.5, digital-(digital-digital.min()+0.5)*(1-frac),
  22                      alpha=0.02, color=PALETTE[1], linewidth=0)
  23 ax2.fill_between(years, digital.min()-0.5, digital, alpha=0.08, color=PALETTE[1], linewidth=0)
  24 
  25 ax1.plot(years, gdp, 'o-', color=PALETTE[0], linewidth=2.2, markersize=6,
  26          label='GDP', markeredgecolor='white', markeredgewidth=1, zorder=5)
  27 ax2.plot(years, digital, 's-', color=PALETTE[1], linewidth=2.2, markersize=6,
  28          label='Digital Index', markeredgecolor='white', markeredgewidth=1, zorder=5)
  29
