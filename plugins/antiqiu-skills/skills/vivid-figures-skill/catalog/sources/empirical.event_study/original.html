1 import numpy as np, matplotlib.pyplot as plt
   2 from _utils.plot_utils import setup_style, save_fig, PALETTE, COLORS, _lighten
   3 setup_style()
   4 
   5 periods = np.arange(-4, 5)
   6 coefs   = np.array([.01, -.02, .01, .00, .12, .25, .31, .38, .42])

   8 
   9 fig, ax = plt.subplots(figsize=(8, 5))

  14 
  15 for t_val, c, s in zip(periods, coefs, se):
  16     lo, hi = c - 1.96 * s, c + 1.96 * s
  17     base_color = COLORS['ref_line'] if t_val < 0 else PALETTE[0]
  18     n_layers = 6
  19     for k in range(n_layers, 0, -1):

  23                 solid_capstyle='round')

  25             solid_capstyle='round', alpha=0.8)

  29         star = '***' if abs(c) > 2.576 * s else '**'
  30         ax.text(t_val, hi + 0.02, star, ha='center', va='bottom',
  31                 fontsize=8, color=base_color, fontweight='bold')
  32 
  33 post_mask = periods >= 0
