1 import numpy as np, matplotlib.pyplot as plt; from _utils.palette_maps import palette_cmap, palette_stops, contrast_text

   4 
   5 models = ['LSTM','GRU','Transformer','ARIMA','Prophet','Ours']
   6 metrics = ['MAE↓','RMSE↓','MAPE(%)↓','R²↑']
   7 data = np.array([[3.21,4.15,5.8,0.921],[3.45,4.38,6.2,0.908],[2.98,3.87,5.1,0.935],
   8                   [5.12,6.73,9.4,0.842],[4.67,5.92,8.1,0.867],[2.45,3.21,4.3,0.952]])
   9 
  10 norm = np.zeros_like(data)

  22

  39 cbar = fig.colorbar(im, ax=ax, shrink=0.7, pad=0.20)
