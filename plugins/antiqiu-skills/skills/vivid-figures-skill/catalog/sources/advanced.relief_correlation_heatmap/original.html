  30 def validate_correlation(matrix, labels):
  31     matrix=np.asarray(matrix,dtype=float)
  32     if (matrix.ndim!=2 or matrix.shape[0]!=matrix.shape[1]
  33             or not 2<=matrix.shape[0]<=40 or len(labels)!=len(matrix)):
  34         raise ValueError('Provide a square correlation matrix of 2–40 labeled variables.')
  35     if len(set(labels))!=len(labels) or any(not str(label).strip() for label in labels):
  36         raise ValueError('Variable labels must be nonempty and unique.')
  37     if not np.isfinite(matrix).all() or np.any(np.abs(matrix)>1+1e-8):
  38         raise ValueError('Correlations must be finite and within [-1,1].')
  39     if not np.allclose(matrix,matrix.T,atol=1e-8,rtol=0) or not np.allclose(np.diag(matrix),1,atol=1e-8,rtol=0):
  40         raise ValueError('Correlation matrices must be symmetric with diagonal 1.')
  41     if np.linalg.eigvalsh(matrix).min() < -1e-6:
  42         raise ValueError('The supplied matrix is not positive semidefinite; verify its source.')
  43     return matrix

  46 def from_samples(samples, labels):
  47     values=np.asarray(samples,dtype=float)
  48     if (values.ndim!=2 or values.shape[0]<3 or values.shape[1]!=len(labels)
  49             or not np.isfinite(values).all() or np.any(values.std(axis=0)==0)):
  50         raise ValueError('Supply at least three complete observations and nonconstant numeric columns.')
  51     return validate_correlation(np.corrcoef(values,rowvar=False),labels)

  54 def cell_geometry(row, col, size, r, relief=1.):
  55     """Return the two side walls and the colored square face, in draw order."""
  56     if not np.isfinite(relief) or relief<0:
  57         raise ValueError('Relief must be finite and nonnegative.')
  58     x,y=float(col),float(size-1-row)
  59     base=np.array([[x,y],[x+1,y],[x+1,y+1],[x,y+1]])
  60     height=relief*(.12+1.4*(float(r)+1)/2)
  61     face=base+height*np.array([.58,.98])
  62     bottom=np.array([base[0],base[1],face[1],face[0]])
  63     left=np.array([base[0],base[3],face[3],face[0]])
  64     return bottom,left,face,height

  67 def plot_chart(matrix, labels, *, cmap=None, relief=1., note=None):
  68     """Deterministic 2.5D oblique projection; full N×N matrix, no smoothing.
  69 
  70     Only square faces use the exact colorbar mapping. Side walls are shaded
  71     for geometry and must not be used to read correlation values.
  72     """
  73     matrix=validate_correlation(matrix,labels)
  74     n=len(labels)
  75     setup_style()
  76     cmap=palette_cmap('diverging') if cmap is None else cmap
  77     norm=Normalize(-1,1)
  78     plt.rcParams.update({'font.family':'sans-serif','font.sans-serif':['Arial','Microsoft YaHei','DejaVu Sans'],
  79                          'pdf.fonttype':42,'axes.unicode_minus':False})
  80     fig=plt.figure(figsize=(11.4,10.2),facecolor='white')
  81     ax=fig.add_axes([.18,.19,.71,.76])
  82     # Rows at the top/back first, columns right to left: nearer cells cover
  83     # hidden parts of farther walls while every complete matrix cell is kept.
  84     faces=[]
  85     for row in range(n):
  86         for col in range(n-1,-1,-1):
  87             r=float(matrix[row,col]);rgb=np.asarray(cmap(norm(r))[:3])
  88             bottom,left,face,height=cell_geometry(row,col,n,r,relief)
  89             for points,color in ((bottom,rgb*.77),(left,rgb*.87),(face,rgb)):
  90                 polygon=Polygon(points,closed=True,facecolor=color,
  91                                 edgecolor='#273641',linewidth=.62,joinstyle='miter')
  92                 ax.add_patch(polygon)
  93             faces.append(dict(row=row,col=col,r=r,height=height,face_color=rgb.tolist()))
  94     max_height=relief*1.52
  95     ax.set_xlim(-.02,n+.58*max_height+.06)
  96     ax.set_ylim(-.02,n+.98*max_height+.06)
  97     ax.set_aspect('equal')
  98     ax.set_xticks(np.arange(n)+.5,labels,rotation=90,fontsize=7.4 if n>16 else 9)
  99     ax.set_yticks(n-np.arange(n)-.5,labels,fontsize=7.4 if n>16 else 9)
 100     ax.tick_params(axis='both',which='both',length=0,pad=3)
 101     ax.grid(False)
 102     for spine in ax.spines.values():spine.set_visible(False)
 103     # Fixed limits and an unshaded colorbar preserve the sign and magnitude.
 104     cax=fig.add_axes([.925,.455,.013,.13])
 105     colorbar=fig.colorbar(ScalarMappable(norm=norm,cmap=cmap),cax=cax,
 106                          ticks=[-1,-.5,0,.5,1])
 107     cax.set_title('Pearson r',loc='left',fontsize=9,fontweight='bold',pad=5)
 108     colorbar.ax.tick_params(labelsize=8,length=2,pad=2)
 109     colorbar.outline.set_visible(False)
 110     if note:fig.text(.18,.04,note,fontsize=8,color='#444444')
 111     return fig,dict(matrix=matrix.tolist(),labels=list(labels),faces=faces,
 112                     height_rule='relief * (0.12 + 1.4 * (r + 1) / 2)',projection=[.58,.98])

 115 def read_csv(path, matrix_input=False):
 116     with Path(path).open(encoding='utf-8-sig',newline='') as stream:rows=list(csv.reader(stream))
 117     if len(rows)<2 or any(len(row)!=len(rows[0]) for row in rows):
 118         raise ValueError('CSV must be rectangular with a header row.')
 119     if matrix_input:
 120         labels=rows[0][1:]
 121         if [r[0] for r in rows[1:]]!=labels:
 122             raise ValueError('Matrix row and column labels and ordering must match.')
 123         matrix=np.array([[float(v) for v in row[1:]] for row in rows[1:]])
 124         return validate_correlation(matrix,labels),labels,None
 125     labels=rows[0];samples=np.array([[float(v) for v in row] for row in rows[1:]])
 126     return from_samples(samples,labels),labels,samples

 129 def main():
 130     parser=argparse.ArgumentParser(description=__doc__)
 131     inputs=parser.add_mutually_exclusive_group()
 132     inputs.add_argument('--data',type=Path,help='Observations CSV; numeric columns with variable names')
 133     inputs.add_argument('--matrix',type=Path,help='Square correlation CSV with row and column labels')
 134     parser.add_argument('--reference-colors',action='store_true',help='Explicit blue reference-style ramp')
 135     parser.add_argument('--relief',type=float,default=1.)
 136     parser.add_argument('--output',type=Path,default=Path('figures/fig_relief_correlation_heatmap'))
 137     args=parser.parse_args()
 138     if args.data or args.matrix:
 139         matrix,labels,samples=read_csv(args.matrix or args.data,matrix_input=args.matrix is not None)
 140         note=None
 141     else:
 142         # Domain labels follow the reference only to demonstrate its dense layout.
 143         # All observations are synthetic latent-factor data, not recovered study data.
 144         labels=['ACs','T-SOD','MPNs','DTPNs','CAT','GSH-PX','Unique AAs','Non-Essential AAs',
 145                 'Essential AAs','Conditionally Essential AAs','SpNDs','Ketone compounds',
 146                 'Hydrocarbon compounds','Alcohol compounds','MDA','T-AOC','Hardness',
 147                 'Springiness','POD','Gumminess','Chewiness','Cohesiveness','Elasticity']
 148         rng=np.random.default_rng(20260916)
 149         latent=rng.normal(size=(360,4))
 150         # Heterogeneous loadings produce varied relief instead of uniform blocks.
 151         loadings=rng.normal(0,.3,size=(23,4))
 152         loadings[:,0]=rng.uniform(.3,1.2,size=23)
 153         loadings[:,1]+=.65*np.sin(np.arange(23)*.85)
 154         loadings[:,2]+=.55*np.cos(np.arange(23)*.6)
 155         loadings[10:15,0]*=-1
 156         noise=rng.uniform(.18,.58,size=23)
 157         samples=latent@loadings.T+rng.normal(size=(360,23))*noise
 158         matrix=from_samples(samples,labels)
 159         note='Synthetic data. Face color = Pearson r; relief increases linearly with r.'
 160     cmap=None
 161     if args.reference_colors:
 162         cmap=LinearSegmentedColormap.from_list('reference_blue',
 163                      ['#F5FCFC','#CFEAF0','#AAC5E4','#6B9FCB','#399DCB'])
 164     fig,stats=plot_chart(matrix,labels,cmap=cmap,relief=args.relief,note=note)
 165     args.output.parent.mkdir(parents=True,exist_ok=True)
 166     for suffix in ('png','pdf'):
 167         fig.savefig(args.output.with_suffix('.'+suffix),dpi=300,facecolor='white',
 168                     bbox_inches='tight',pad_inches=.12)
 169     args.output.with_suffix('.json').write_text(json.dumps(stats,ensure_ascii=False,indent=2),encoding='utf8')
 170     with args.output.with_suffix('.csv').open('w',encoding='utf8',newline='') as stream:
 171         writer=csv.writer(stream);writer.writerow(['variable',*labels])
 172         writer.writerows([label,*row] for label,row in zip(labels,matrix))
 173     if args.data is None and args.matrix is None:
 174         with args.output.with_name(args.output.name+'-samples.csv').open('w',encoding='utf8',newline='') as stream:
 175             writer=csv.writer(stream);writer.writerow(labels);writer.writerows(samples)
 176     plt.close(fig)
 177     print('Saved PNG, vector PDF, matrix CSV and geometry:',args.output)
