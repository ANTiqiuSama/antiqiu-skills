1 from _utils.plot_utils import setup_style, save_fig, PALETTE, COLORS, _lighten, smart_labels
   2 setup_style()
   3 import matplotlib.pyplot as plt

   8 methods = ['Ours', 'Baseline-A', 'Baseline-B', 'Baseline-C', 'Baseline-D']
   9 scores = [0.923, 0.887, 0.862, 0.841, 0.815]
  10 
  11 n = len(methods)
  12 score_min, score_max = min(scores), max(scores)
  13 score_range = score_max - score_min if score_max > score_min else 1
  14 
  15 # ── 渐变配色：从紫罗兰 → 珊瑚橙（明度和色相同时渐变）
  16 color_top = '#7B6BA5'     # 紫罗兰（高分，偏冷一点）
  17 color_bottom = '#E08B74'  # 珊瑚橙（低分，偏暖）
  18 
  19 def interpolate_color(c1, c2, t):
  20     """HSL 空间插值：t=0 返回 c1，t=1 返回 c2"""
  21     r1, g1, b1 = mc.to_rgb(c1)
  22     r2, g2, b2 = mc.to_rgb(c2)
  23     h1, l1, s1 = colorsys.rgb_to_hls(r1, g1, b1)
  24     h2, l2, s2 = colorsys.rgb_to_hls(r2, g2, b2)
  25     if abs(h2 - h1) > 0.5:
  26         if h1 < h2: h1 += 1.0
  27         else: h2 += 1.0
  28     h = (h1 + (h2 - h1) * t) % 1.0
  29     l = l1 + (l2 - l1) * t
  30     s = s1 + (s2 - s1) * t
  31     return colorsys.hls_to_rgb(h, l, s)
  32 
  33 item_colors = [interpolate_color(color_top, color_bottom, i / (n - 1) if n > 1 else 0) for i in range(n)]
  34 
  35 # ── 自适应高度（每项 0.46 高度 + 上下留白）
  36 _fig_h = max(4, n * 0.46 + 1.8)

  38 y_pos = np.arange(n)

  80         ax.text(badge_x, y_pos[i], str(rank),
