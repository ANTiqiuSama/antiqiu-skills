  16 DEFAULT_ORDER = ["USE", "ATT", "PBC", "BI", "EVA", "AWA", "SN", "ETH"]

  29 def load_data(path: Path) -> pd.DataFrame:
  30     suffix = path.suffix.lower()
  31     if suffix in {".xlsx", ".xls"}:
  32         return pd.read_excel(path)
  33     if suffix == ".sav":
  34         try:
  35             import pyreadstat
  36         except ImportError as exc:
  37             raise SystemExit("Reading .sav requires pyreadstat. Install it or use an XLSX source.") from exc
  38         df, _meta = pyreadstat.read_sav(str(path), apply_value_formats=False)
  39         return df
  40     if suffix == ".csv":
  41         return pd.read_csv(path)
  42     raise SystemExit(f"Unsupported data format: {path.suffix}")

  45 def parse_json_arg(value: str | None, default):
  46     if not value:
  47         return default
  48     candidate = Path(value)
  49     try:
  50         if candidate.exists():
  51             return json.loads(candidate.read_text(encoding="utf-8"))
  52     except OSError:
  53         pass
  54     return json.loads(value)

  57 def auto_construct_map(columns: list[str], exclude: dict[str, list[int]]) -> dict[str, list[str]]:
  58     mapping: dict[str, list[str]] = {}
  59     for construct, prefixes in PREFIX_GROUPS.items():
  60         selected = [c for c in columns if str(c).startswith(prefixes)]
  61         if selected:
  62             removed = set(exclude.get(construct, []))
  63             keep = [i for i in range(len(selected)) if i not in removed]
  64             mapping[construct] = [selected[i] for i in keep]
  65     return mapping

  68 def build_scores(df: pd.DataFrame, mapping: dict[str, list[str]], order: list[str]) -> pd.DataFrame:
  69     missing = [c for cols in mapping.values() for c in cols if c not in df.columns]
  70     if missing:
  71         raise SystemExit(f"Construct map references missing columns: {missing[:8]}")
  72     scored = pd.DataFrame(index=df.index)
  73     for construct in order:
  74         cols = mapping.get(construct)
  75         if not cols:
  76             raise SystemExit(f"No item mapping found for construct: {construct}")
  77         scored[construct] = df[cols].apply(pd.to_numeric, errors="coerce").mean(axis=1)
  78     return scored

  81 def standardize(df: pd.DataFrame) -> pd.DataFrame:
  82     return (df - df.mean()) / df.std(ddof=0)

  85 def residualize_each_variable(scored: pd.DataFrame, order: list[str]) -> pd.DataFrame:
  86     residuals = pd.DataFrame(index=scored.index)
  87     for y in order:
  88         x_cols = [c for c in order if c != y]
  89         x = scored[x_cols].to_numpy(float)
  90         yv = scored[y].to_numpy(float)
  91         mask = np.isfinite(yv) & np.isfinite(x).all(axis=1)
  92         X = np.column_stack([np.ones(mask.sum()), x[mask]])
  93         beta, *_ = np.linalg.lstsq(X, yv[mask], rcond=None)
  94         residuals.loc[mask, y] = yv[mask] - (X @ beta)
  95     return standardize(residuals[order])

  98 def p_values(scored: pd.DataFrame, order: list[str]) -> pd.DataFrame:
  99     pmat = pd.DataFrame(index=order, columns=order, dtype=float)
 100     for a in order:
 101         for b in order:
 102             if a == b:
 103                 pmat.loc[a, b] = 0.0
 104                 continue
 105             pair = scored[[a, b]].dropna()
 106             _r, p = stats.pearsonr(pair[a], pair[b])
 107             pmat.loc[a, b] = p
 108     return pmat

 111 def star(p: float) -> str:
 112     if p < 0.001:
 113         return "***"
 114     if p < 0.01:
 115         return "**"
 116     if p < 0.05:
 117         return "*"
 118     return ""

 130 def paper_comparison(corr: pd.DataFrame, paper: dict | None, order: list[str]) -> pd.DataFrame:
 131     if not paper:
 132         return pd.DataFrame()
 133     rows = []
 134     for a, b in combinations(order, 2):
 135         target = None
 136         if isinstance(paper.get(a), dict) and b in paper[a]:
 137             target = paper[a][b]
 138         elif isinstance(paper.get(b), dict) and a in paper[b]:
 139             target = paper[b][a]
 140         elif f"{a},{b}" in paper:
 141             target = paper[f"{a},{b}"]
 142         elif f"{b},{a}" in paper:
 143             target = paper[f"{b},{a}"]
 144         if target is None:
 145             continue
 146         rows.append({
 147             "var1": a,
 148             "var2": b,
 149             "computed": float(corr.loc[a, b]),
 150             "paper": float(target),
 151             "abs_diff": abs(float(corr.loc[a, b]) - float(target)),
 152         })
 153     return pd.DataFrame(rows).sort_values("abs_diff", ascending=False)

 156 def draw_plots(
 157     scored: pd.DataFrame,
 158     residuals: pd.DataFrame,
 159     corr: pd.DataFrame,
 160     pmat: pd.DataFrame,
 161     order: list[str],
 162     out_dir: Path,
 163     prefix: str,
 164 ) -> dict[str, str]:
 165     import matplotlib
 166     matplotlib.use("Agg")
 167     import matplotlib.pyplot as plt
 168     from matplotlib import font_manager
 169     from matplotlib.cm import ScalarMappable
 170     from matplotlib.colors import LinearSegmentedColormap, Normalize
 171     from matplotlib.ticker import FuncFormatter
 172     import seaborn as sns
 173     from PIL import Image
 174 
 175     font_path = available_times_font()
 176     if font_path:
 177         font_manager.fontManager.addfont(font_path)
 178     plt.rcParams.update({
 179         "font.family": "Times New Roman",
 180         "font.size": 24,
 181         "axes.labelsize": 30,
 182         "xtick.labelsize": 24,
 183         "ytick.labelsize": 24,
 184         "axes.unicode_minus": False,
 185         "figure.dpi": 180,
 186         "savefig.dpi": 300,
 187     })
 188     sns.set_theme(style="whitegrid", rc={
 189         "font.family": "Times New Roman",
 190         "axes.edgecolor": "#AEB7C4",
 191         "grid.color": "#E5EAF0",
 192         "axes.facecolor": "#FFFFFF",
 193         "figure.facecolor": "#FFFFFF",
 194     })
 195 
 196     fill = "#87DED7"
 197     line = "#3049A1"
 198     dot = "#2D84AD"
 199     cmap = LinearSegmentedColormap.from_list(
 200         "paper_teal_blue",
 201         ["#25258F", "#5A69A8", "#F6FAFA", "#BEEAD6", "#53CDC9", "#4A98B9", "#25258F"],
 202         N=256,
 203     )
 204 
 205     long = residuals.melt(var_name="Variable", value_name="Residual")
 206     long["Variable"] = pd.Categorical(long["Variable"], categories=order, ordered=True)
 207     fig, ax = plt.subplots(figsize=(6.4, 11.0))
 208     sns.violinplot(
 209         data=long,
 210         y="Variable",
 211         x="Residual",
 212         order=order,
 213         orient="h",
 214         inner=None,
 215         color=fill,
 216         linewidth=1.8,
 217         cut=0,
 218         ax=ax,
 219     )
 220     for collection in ax.collections:
 221         collection.set_edgecolor(line)
 222         collection.set_alpha(0.75)
 223     rng = np.random.default_rng(20260626)
 224     for yi, var in enumerate(order):
 225         vals = long.loc[long["Variable"] == var, "Residual"].dropna().to_numpy()
 226         ax.scatter(vals, rng.normal(yi, 0.055, len(vals)), s=11, color=dot, alpha=0.28, edgecolors="none", zorder=3)
 227     ax.axvline(0, color="#6E7785", linestyle=":", linewidth=1.4)
 228     ax.set_xlim(-2.8, 2.8)
 229     ax.set_xlabel("Residual", fontsize=30)
 230     ax.set_ylabel("")
 231     ax.tick_params(axis="y", labelsize=30)
 232     ax.tick_params(axis="x", labelsize=26)
 233     ax.grid(axis="x", color="#E5EAF0", linewidth=1)
 234     ax.grid(axis="y", visible=False)
 235     sns.despine(ax=ax, top=False, right=False)
 236     ax.text(-0.02, 1.02, "(a)", transform=ax.transAxes, fontsize=34, fontweight="bold", ha="left", va="bottom")
 237     fig.tight_layout(pad=1.2)
 238     violin_png = out_dir / f"{prefix}_residual_violin.png"
 239     violin_svg = out_dir / f"{prefix}_residual_violin.svg"
 240     fig.savefig(violin_png, bbox_inches="tight")
 241     fig.savefig(violin_svg, bbox_inches="tight")
 242     plt.close(fig)
 243 
 244     n = len(order)
 245     fig = plt.figure(figsize=(16.8, 14.4))
 246     gs = fig.add_gridspec(n, n, wspace=0.03, hspace=0.03)
 247     norm = Normalize(vmin=-1, vmax=1)
 248     for i, row in enumerate(order):
 249         for j, col in enumerate(order):
 250             ax = fig.add_subplot(gs[i, j])
 251             for spine in ax.spines.values():
 252                 spine.set_color("#AEB7C4")
 253                 spine.set_linewidth(1.0)
 254             ax.tick_params(length=0, pad=3)
 255             if i == j:
 256                 vals = scored[row].dropna()
 257                 ax.hist(vals, bins=np.arange(1, 8.5, 0.5), color="#CBEFC9", edgecolor="#6FAB74", linewidth=1.0)
 258                 try:
 259                     sns.kdeplot(x=vals, ax=ax, color=line, linewidth=1.6, bw_adjust=0.9)
 260                 except Exception:
 261                     pass
 262                 ax.set_xlim(0.8, 7.2)
 263                 ax.set_yticks([])
 264                 ax.set_xlabel("")
 265                 ax.set_ylabel("")
 266             elif i > j:
 267                 ax.scatter(scored[col], scored[row], s=10, color="#2BB8C6", alpha=0.34, edgecolors="none")
 268                 ax.set_xlim(0.8, 7.2)
 269                 ax.set_ylim(0.8, 7.2)
 270             else:
 271                 value = corr.loc[row, col]
 272                 ax.set_facecolor(cmap(norm(value)))
 273                 ax.text(0.5, 0.57, f"{value:.3f}", transform=ax.transAxes, ha="center", va="center", fontsize=24, color="#202633")
 274                 ax.text(
 275                     0.5,
 276                     0.38,
 277                     star(float(pmat.loc[row, col])),
 278                     transform=ax.transAxes,
 279                     ha="center",
 280                     va="center",
 281                     fontsize=24,
 282                     fontweight="bold",
 283                     color="#202633",
 284                 )
 285                 ax.set_xticks([])
 286                 ax.set_yticks([])
 287             if i < n - 1:
 288                 ax.set_xticklabels([])
 289             else:
 290                 ax.set_xlabel(col, labelpad=10, fontsize=30)
 291             if j > 0:
 292                 ax.set_yticklabels([])
 293             else:
 294                 ax.set_ylabel(row, rotation=90, labelpad=20, fontsize=30)
 295             ax.grid(False)
 296     fig.text(0.03, 0.965, "(b)", fontsize=34, fontweight="bold", ha="left", va="top")
 297     cax = fig.add_axes([0.92, 0.13, 0.025, 0.75])
 298     cb = fig.colorbar(ScalarMappable(norm=norm, cmap=cmap), cax=cax)
 299     cb.set_ticks([-1, -0.75, -0.5, -0.25, 0, 0.25, 0.5, 0.75, 1])
 300     cb.ax.yaxis.set_major_formatter(FuncFormatter(lambda x, _pos: "0.00" if abs(x) < 0.0005 else f"{x:.2f}"))
 301     cb.ax.tick_params(labelsize=24)
 302     cb.outline.set_edgecolor("#AEB7C4")
 303     fig.subplots_adjust(left=0.08, right=0.89, top=0.93, bottom=0.08)
 304     pearson_png = out_dir / f"{prefix}_pearson_pairgrid.png"
 305     pearson_svg = out_dir / f"{prefix}_pearson_pairgrid.svg"
 306     fig.savefig(pearson_png, bbox_inches="tight")
 307     fig.savefig(pearson_svg, bbox_inches="tight")
 308     plt.close(fig)
 309 
 310     img_a = Image.open(violin_png).convert("RGB")
 311     img_b = Image.open(pearson_png).convert("RGB")
 312     h = max(img_a.height, img_b.height)
 313     resample = getattr(Image, "Resampling", Image).LANCZOS
 314     img_a = img_a.resize((round(img_a.width * h / img_a.height), h), resample)
 315     img_b = img_b.resize((round(img_b.width * h / img_b.height), h), resample)
 316     combined = Image.new("RGB", (img_a.width + img_b.width + 40, h), "white")
 317     combined.paste(img_a, (0, 0))
 318     combined.paste(img_b, (img_a.width + 40, 0))
 319     combined_png = out_dir / f"{prefix}_violin_pearson_combined.png"
 320     combined.save(combined_png, quality=95)
 321     return {
 322         "violin_png": str(violin_png),
 323         "violin_svg": str(violin_svg),
 324         "pearson_png": str(pearson_png),
 325         "pearson_svg": str(pearson_svg),
 326         "combined_png": str(combined_png),
 327     }

 330 def write_audit(
 331     out_dir: Path,
 332     prefix: str,
 333     mapping: dict[str, list[str]],
 334     scored: pd.DataFrame,
 335     residuals: pd.DataFrame,
 336     corr: pd.DataFrame,
 337     comparison: pd.DataFrame,
 338 ) -> Path:
 339     out = out_dir / f"{prefix}_plot_audit.xlsx"
 340     rows = [{"construct": k, "items_used": " | ".join(map(str, mapping[k]))} for k in mapping]
 341     with pd.ExcelWriter(out, engine="openpyxl") as writer:
 342         pd.DataFrame(rows).to_excel(writer, sheet_name="item_mapping", index=False)
 343         scored.to_excel(writer, sheet_name="construct_scores", index=False)
 344         residuals.to_excel(writer, sheet_name="residuals_z", index=False)
 345         corr.round(6).to_excel(writer, sheet_name="computed_corr")
 346         if not comparison.empty:
 347             comparison.round(6).to_excel(writer, sheet_name="paper_comparison", index=False)
 348     return out

 351 def main() -> None:
 352     parser = argparse.ArgumentParser()
 353     parser.add_argument("--data", required=True, type=Path)
 354     parser.add_argument("--output-dir", required=True, type=Path)
 355     parser.add_argument("--order", default=",".join(DEFAULT_ORDER))
 356     parser.add_argument("--exclude-json", default=None, help='JSON such as {"AWA":[0,2]} using zero-based item positions after auto-detection.')
 357     parser.add_argument("--construct-map-json", default=None, help="JSON string or file path mapping constructs to exact column names.")
 358     parser.add_argument("--paper-corr-json", default=None, help="Optional JSON string or file path with paper correlations for comparison.")
 359     parser.add_argument("--prefix", default="sem")
 360     args = parser.parse_args()
 361 
 362     out_dir = args.output_dir
 363     out_dir.mkdir(parents=True, exist_ok=True)
 364     order = [x.strip() for x in args.order.split(",") if x.strip()]
 365     exclude = parse_json_arg(args.exclude_json, {})
 366     data = load_data(args.data)
 367     mapping = parse_json_arg(args.construct_map_json, None) or auto_construct_map(list(data.columns), exclude)
 368     scored = build_scores(data, mapping, order)
 369     corr = scored[order].corr()
 370     pmat = p_values(scored, order)
 371     residuals = residualize_each_variable(scored, order)
 372     paper = parse_json_arg(args.paper_corr_json, None)
 373     comparison = paper_comparison(corr, paper, order)
 374     figures = draw_plots(scored, residuals, corr, pmat, order, out_dir, args.prefix)
 375     audit = write_audit(out_dir, args.prefix, mapping, scored[order], residuals[order], corr, comparison)
 376     summary = {
 377         "source_data": str(args.data),
 378         "n_rows": int(data.shape[0]),
 379         "n_complete_construct_rows": int(scored[order].dropna().shape[0]),
 380         "order": order,
 381         "mapping": mapping,
 382         "figures": figures,
 383         "audit_workbook": str(audit),
 384     }
 385     if not comparison.empty:
 386         summary["mean_abs_diff_vs_paper_corr"] = float(comparison["abs_diff"].mean())
 387         summary["max_abs_diff_vs_paper_corr"] = float(comparison["abs_diff"].max())
 388     summary_path = out_dir / f"{args.prefix}_plot_summary.json"
 389     summary_path.write_text(json.dumps(summary, ensure_ascii=True, indent=2), encoding="utf-8")
 390     print(json.dumps(summary, ensure_ascii=True, indent=2))
