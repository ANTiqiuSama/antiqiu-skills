1 import numpy as np
   2 import matplotlib.pyplot as plt
   3 from _utils.plot_utils import setup_style, save_fig, PALETTE, COLORS, _lighten
   4 setup_style()
   5 
   6 np.random.seed(42)
   7 fig, axes = plt.subplots(2, 2, figsize=(5.0, 4.9))   # 示例尺寸；保留模板比例，按实际显示尺寸检查字号与布局
   8 
   9 # (a) 折线图 — 渐变填充
  10 ax = axes[0, 0]
  11 for i in range(3):
  12     y = np.cumsum(np.random.randn(10))
  13     ax.plot(range(10), y, 'o-', color=PALETTE[i], linewidth=1.8, markersize=5,
  14             markeredgecolor='white', markeredgewidth=0.8, label=f'系列{i + 1}')
  15     ax.fill_between(range(10), y - 1, y + 1, alpha=0.08, color=PALETTE[i])
  16 ax.legend(frameon=False, labelspacing=0.35, handlelength=1.6, fontsize=8)
  17 ax.set_ylabel('累计值', fontsize=10)
  18 ax.grid(alpha=0.12, linestyle='--', color=COLORS['grid'])
  19 ax.spines['top'].set_visible(False)
  20 ax.spines['right'].set_visible(False)
  21 
  22 # (b) 分组柱状图 — 淡色填充 + 原色边框

  26     vals = np.random.uniform(60, 95, 5)

  29            edgecolor=PALETTE[i], linewidth=1.2, label=label)
  30 ax.legend(frameon=False, labelspacing=0.35, handlelength=1.6, fontsize=8)
  31 ax.set_ylabel('得分', fontsize=10)
  32 ax.grid(axis='y', alpha=0.12, linestyle='--', color=COLORS['grid'])
  33 ax.spines['top'].set_visible(False)
  34 ax.spines['right'].set_visible(False)
  35 
  36 # (c) 散点图 — KDE 等高线背景
  37 ax = axes[1, 0]

  41                edgecolor='white', linewidth=0.5, label=f'类{i + 1}')
  42 ax.legend(frameon=False, labelspacing=0.35, handlelength=1.6, fontsize=8)
