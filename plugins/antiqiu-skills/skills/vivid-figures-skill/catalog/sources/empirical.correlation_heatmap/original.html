1 import numpy as np, matplotlib.pyplot as plt, seaborn as sns; from _utils.palette_maps import palette_cmap, palette_stops, contrast_text
   2 from scipy.cluster.hierarchy import linkage, dendrogram
   3 from scipy.stats import pearsonr
   4 from _utils.plot_utils import setup_style, save_fig, COLORS, _lighten
   5 setup_style()
   6 
   7 np.random.seed(42)
   8 labels = ['GDP', 'Digital', 'HumanCap', 'Urban', 'FDI', 'R&D']
   9 n_vars = len(labels)
  10 raw = np.random.randn(n_vars, 100)
  11 raw[1] += 0.6*raw[0]; raw[2] += 0.4*raw[0]; raw[4] += 0.5*raw[3]; raw[5] += 0.7*raw[1]
  12 corr = np.corrcoef(raw)
  13 
  14 pvals = np.zeros((n_vars, n_vars))
  15 for i in range(n_vars):
  16     for j in range(n_vars):
  17         if i != j: _, pvals[i, j] = pearsonr(raw[i], raw[j])
  18 
  19 Z = linkage(1 - np.abs(corr), method='ward')
  20 fig = plt.figure(figsize=(10, 9))
  21 gs = fig.add_gridspec(2, 1, height_ratios=[1, 7], hspace=0.02)
  22 
  23 # 顶部树状图（只保留顶部，不用左侧，避免挡标签）
  24 ax_dendro = fig.add_subplot(gs[0])
  25 dendrogram(Z, labels=labels, ax=ax_dendro, leaf_rotation=0, color_threshold=0, above_threshold_color=COLORS['ref_line'])
  26 ax_dendro.set_xticks([]); ax_dendro.spines[:].set_visible(False)
  27 ax_dendro.tick_params(left=False, labelleft=False, bottom=False)
  28 
  29 ax_heat = fig.add_subplot(gs[1])
  30 mask = np.triu(np.ones_like(corr, dtype=bool), k=1)
  31 sns.heatmap(corr, mask=mask, annot=False, cmap=palette_cmap('diverging'), center=0, square=True,
  32             linewidths=1.0, linecolor='white', xticklabels=labels, yticklabels=labels,
  33             cbar_kws={'shrink': 0.7, 'label': 'Correlation'}, ax=ax_heat, vmin=-1, vmax=1)
  34 
  35 for i in range(n_vars):
  36     for j in range(n_vars):
  37         if j <= i:
  38             val = corr[i, j]
  39             p = pvals[i, j] if i != j else 0
  40             stars = '***' if p < 0.001 else ('**' if p < 0.01 else ('*' if p < 0.05 else ''))
  41             txt_color = contrast_text(ax_heat.collections[0].cmap(ax_heat.collections[0].norm(val)))
  42             ax_heat.text(j + 0.5, i + 0.5, f'{val:.2f}{stars}', ha='center', va='center',
  43                          fontsize=8.5, color=txt_color, fontweight='bold' if i == j else 'normal')
  44 
  45 fig.tight_layout()
  46 save_fig(fig, 'figures/fig_corr.pdf')
