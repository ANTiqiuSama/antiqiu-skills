1 from _utils.plot_utils import setup_style, save_fig, PALETTE, COLORS, _lighten; from _utils.palette_maps import palette_cmap, palette_stops, contrast_text
   2 setup_style()
   3 import matplotlib.pyplot as plt
   4 import numpy as np
   5 from scipy.cluster.hierarchy import dendrogram, linkage
   6 from scipy.spatial.distance import pdist
   7 
   8 methods = ['Ours', 'LSTM', 'Random Forest', 'Linear Reg']
   9 metrics = ['MAE', 'RMSE', 'R2', 'MAPE(%)', 'Speed']
  10 # Lower is better for MAE/RMSE/MAPE, higher for R2/Speed
  11 data = np.array([
  12     [0.29, 1.05, 0.987, 2.1, 0.85],
  13     [72.23, 85.4, 0.812, 15.3, 0.60],
  14     [0.45, 1.82, 0.965, 3.8, 0.92],
  15     [137.08, 152.3, 0.421, 48.2, 0.98],
  16 ])
  17 higher_better = [False, False, True, False, True]
  18 
  19 # Rank medals
  20 rank_symbols = ['🥇', '🥈', '🥉', '④']
  21 
  22 # Normalize for color mapping
  23 norm_data = np.zeros_like(data)
  24 for j in range(data.shape[1]):
  25     col = data[:, j]
  26     if higher_better[j]:
  27         norm_data[:, j] = (col - col.min()) / (col.max() - col.min() + 1e-10)
  28     else:
  29         norm_data[:, j] = 1 - (col - col.min()) / (col.max() - col.min() + 1e-10)
  30 
  31 # Compute overall ranking (average normalized rank)
  32 overall_rank_score = np.mean(norm_data, axis=1)
  33 
  34 # Add overall ranking column
  35 metrics_ext = metrics + ['Overall']
  36 data_ext = np.column_stack([data, overall_rank_score])
  37 norm_ext = np.column_stack([norm_data, overall_rank_score / overall_rank_score.max()])
  38 
  39 fig = plt.figure(figsize=(9, 4.5))
  40 
  41 # Dendrogram on top
  42 ax_dendro = fig.add_axes([0.15, 0.82, 0.65, 0.14])
  43 Z = linkage(pdist(norm_data), method='ward')
  44 dn = dendrogram(Z, labels=methods, ax=ax_dendro, leaf_font_size=0,
  45                 color_threshold=0, above_threshold_color=COLORS['ref_line'])
  46 ax_dendro.set_axis_off()
  47 row_order = dn['leaves']
  48 
  49 # Heatmap
  50 ax = fig.add_axes([0.15, 0.12, 0.72, 0.68])
  51 ordered_norm = norm_ext[row_order]
  52 ordered_data = data_ext[row_order]
  53 ordered_methods = [methods[i] for i in row_order]
  54 
  55 im = ax.imshow(ordered_norm, cmap=palette_cmap('sequential'), aspect='auto', vmin=0, vmax=1)
  56 ax.set_xticks(range(len(metrics_ext)))
  57 ax.set_xticklabels(metrics_ext, fontsize=10)
  58 ax.set_yticks(range(len(methods)))
  59 ax.set_yticklabels(ordered_methods, fontsize=10)
  60 
  61 # Annotate with values, ranks, and 浅色填充+原色边框 for best
  62 for j in range(len(metrics_ext)):
  63     if j < len(metrics):
  64         col_vals = data[:, j]
  65         if higher_better[j]:
  66             rank_order = np.argsort(-col_vals)
  67         else:
  68             rank_order = np.argsort(col_vals)
  69     else:
  70         rank_order = np.argsort(-overall_rank_score)
  71 
  72     for i_orig, rank_pos in enumerate(rank_order):
  73         # Find position in ordered display
  74         i_display = row_order.index(rank_pos)
  75         if j < len(metrics):
  76             val_str = f'{data[rank_pos, j]:.2f}'
  77         else:
  78             val_str = f'{overall_rank_score[rank_pos]:.2f}'
  79 
  80         rank_idx = i_orig
  81         rank_label = rank_symbols[rank_idx] if rank_idx < len(rank_symbols) else ''
  82 
  83         color = contrast_text(im.cmap(im.norm(ordered_norm[i_display, j])))
  84         weight = 'bold' if rank_idx == 0 else 'normal'
  85 
  86         ax.text(j, i_display, f'{val_str}\n{rank_label}', ha='center', va='center',
  87                 fontsize=8.5, fontweight=weight, color=color)
  88 
  89         # 列最优：浅色填充背景 + 原色粗边框
  90         if rank_idx == 0:
  91             rect = plt.Rectangle((j - 0.5, i_display - 0.5), 1, 1,
  92                                   linewidth=2.5, edgecolor=PALETTE[0],
  93                                   facecolor=_lighten(PALETTE[0], 0.5),
  94                                   alpha=0.3, zorder=4)
  95             ax.add_patch(rect)
  96             # 原色边框（不透明）
  97             rect_border = plt.Rectangle((j - 0.5, i_display - 0.5), 1, 1,
  98                                          linewidth=2.5, edgecolor=PALETTE[0],
  99                                          facecolor='none', zorder=5)
 100             ax.add_patch(rect_border)
 101 
 102 ax.spines[:].set_visible(False)
 103 
 104 # Colorbar
 105 ax_cbar = fig.add_axes([0.89, 0.12, 0.02, 0.68])
 106 cbar = plt.colorbar(im, cax=ax_cbar)
 107 cbar.set_label('Normalized (1=best)', fontsize=8)
 108 cbar.ax.tick_params(labelsize=7)
 109 
 110 save_fig(fig, 'figures/fig_method_heatmap.pdf')
