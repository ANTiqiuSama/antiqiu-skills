1 import numpy as np
   2 import matplotlib.pyplot as plt
   3 from scipy.stats import gaussian_kde
   4 from _utils.plot_utils import setup_style, save_fig, PALETTE, COLORS, _lighten
   5 setup_style()
   6 
   7 np.random.seed(42)
   8 groups = ['Model A', 'Model B', 'Model C', 'Model D', 'Model E',
   9           'Model F', 'Model G', 'Model H']
  10 n_groups = len(groups)
  11 
  12 # Generate sample data (replace with real data)
  13 data = []
  14 for i in range(n_groups):
  15     center = 70 + i * 3 + np.random.randn() * 2
  16     spread = 5 + np.random.rand() * 5
  17     d = np.random.normal(center, spread, 200)
  18     data.append(d)
  19 
  20 fig, ax = plt.subplots(figsize=(8, 6))
  21 overlap = 0.6  # vertical overlap factor
  22 x_grid = np.linspace(min(d.min() for d in data) - 5,
  23                       max(d.max() for d in data) + 5, 300)
  24 
  25 for i in range(n_groups - 1, -1, -1):  # draw back to front
  26     kde = gaussian_kde(data[i], bw_method=0.3)
  27     density = kde(x_grid)
  28     # Normalize density to consistent height
  29     density = density / density.max() * 0.8
  30 
  31     baseline = i * overlap
  32     color = PALETTE[i % len(PALETTE)]
  33     light = _lighten(color, 0.5)
  34 
  35     # Gradient fill
  36     ax.fill_between(x_grid, baseline, baseline + density,
  37                     color=light, alpha=0.85, zorder=n_groups - i)
  38     ax.plot(x_grid, baseline + density, color=color, linewidth=1.5,
  39             zorder=n_groups - i + 0.5)
  40 
  41     # Median line
  42     median = np.median(data[i])
  43     med_density = kde(median)[0] / density.max() * 0.8
  44     ax.plot([median, median], [baseline, baseline + med_density],
  45             color=color, linewidth=1.5, linestyle='--', alpha=0.7,
  46             zorder=n_groups - i + 1)
  47     ax.text(median, baseline + med_density + 0.02,
  48             f'{median:.1f}', ha='center', fontsize=7, color=color,
  49             fontweight='bold', zorder=n_groups + 10)
  50 
  51     # Group label
  52     ax.text(x_grid[0] - 1, baseline + 0.15, groups[i],
  53             ha='right', va='center', fontsize=9, fontweight='bold',
  54             color=color)
  55 
  56 ax.set_yticks([])
  57 ax.set_xlabel('Score', fontsize=11)
  58 ax.spines['left'].set_visible(False)
  59 ax.spines['top'].set_visible(False)
  60 ax.spines['right'].set_visible(False)
  61 fig.tight_layout()
  62 save_fig(fig, 'figures/fig_ridgeline.pdf')
