  16 G = nx.karate_club_graph()

  19 # Assign community colors

  22     for node in comm:

  26 # Node sizing by degree

  31 fig, ax = plt.subplots(figsize=(8, 7))
  32 pos = nx.spring_layout(G, seed=42, k=0.5)
  33 
  34 # Draw convex hulls for communities
  35 for i, comm in enumerate(communities):
  36     if len(comm) >= 3:
  37         points = np.array([pos[n] for n in comm])
  38         try:
  39             hull = ConvexHull(points)
  40             hull_points = points[hull.vertices]
  41             # Close the polygon
  42             hull_points = np.vstack([hull_points, hull_points[0]])
  43             # Expand hull slightly
  44             centroid = points.mean(axis=0)
  45             expanded = centroid + 1.15 * (hull_points - centroid)
  46             ax.fill(expanded[:, 0], expanded[:, 1],
  47                     color=PALETTE[i % len(PALETTE)], alpha=0.08)
  48             ax.plot(expanded[:, 0], expanded[:, 1],
  49                     color=PALETTE[i % len(PALETTE)], linewidth=1.5,
  50                     linestyle='--', alpha=0.4)
  51         except Exception:
  52             pass
  53 
  54 # Edge weight gradient
  55 edges = G.edges()

  57 max_w = max(edge_weights) if edge_weights else 1
  58 cmap_edge = mcolors.LinearSegmentedColormap.from_list('ew', [_lighten(scale_colors[0], 0.8), COLORS['ref_line']])
  59 for (u, v), w in zip(edges, edge_weights):
