  77     # ===== 方案 B：纯 matplotlib（不依赖 geopandas，同样有省份轮廓） =====
  78     def _extract_polygons(geometry):
  79         """从 GeoJSON geometry 提取所有 polygon 坐标环"""
  80         polys = []
  81         gtype = geometry['type']
  82         if gtype == 'Polygon':
  83             for ring in geometry['coordinates']:
  84                 polys.append(np.array(ring))
  85         elif gtype == 'MultiPolygon':
  86             for polygon in geometry['coordinates']:
  87                 for ring in polygon:
  88                     polys.append(np.array(ring))
  89         return polys
  90
