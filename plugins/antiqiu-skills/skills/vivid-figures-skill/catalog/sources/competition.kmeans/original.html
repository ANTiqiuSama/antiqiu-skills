1 import numpy as np, matplotlib.pyplot as plt
   2 from scipy.stats import gaussian_kde
   3 from sklearn.datasets import make_blobs
   4 from _utils.plot_utils import setup_style, save_fig, PALETTE, COLORS
   5 setup_style()
   6 np.random.seed(42)
   7 
   8 X, y_true = make_blobs(n_samples=300, centers=4, cluster_std=1.0)
   9 from sklearn.cluster import KMeans
  10 km = KMeans(n_clusters=4, random_state=42, n_init=10).fit(X)
  11 labels = km.labels_
  12 centers = km.cluster_centers_
  13 
  14 fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5))
  15 
  16 # 左图：聚类散点
  17 for k in range(4):
  18     mask = labels == k
  19     ax1.scatter(X[mask, 0], X[mask, 1], s=20, alpha=0.5, color=PALETTE[k],
  20                edgecolor='white', linewidth=0.3, label=f'簇 {k+1}')
  21 ax1.scatter(centers[:, 0], centers[:, 1], s=200, color=COLORS['text'], marker='X',
  22            edgecolor='white', linewidth=2, zorder=5, label='质心')
  23 ax1.set_xlabel('特征维度 1', fontsize=11); ax1.set_ylabel('特征维度 2', fontsize=11)
  24 ax1.legend(frameon=False, labelspacing=0.35, handlelength=1.6, fontsize=8, loc='best')
  25 ax1.spines['top'].set_visible(False); ax1.spines['right'].set_visible(False)
  26 
  27 # 右图：轮廓系数
  28 from sklearn.metrics import silhouette_samples
  29 sil = silhouette_samples(X, labels)
  30 y_lower = 0
  31 for k in range(4):
  32     cluster_sil = np.sort(sil[labels == k])
  33     y_upper = y_lower + len(cluster_sil)
  34     ax2.fill_betweenx(np.arange(y_lower, y_upper), 0, cluster_sil,
  35                        alpha=0.5, color=PALETTE[k])
  36     ax2.text(-0.05, y_lower + 0.5 * len(cluster_sil), f'簇{k+1}',
  37             fontsize=9, fontweight='bold', color=PALETTE[k])
  38     y_lower = y_upper + 5
  39 avg_sil = np.mean(sil)
  40 ax2.axvline(avg_sil, color=COLORS['down'], linewidth=1.5, linestyle='--',
  41            label=f'平均轮廓系数={avg_sil:.3f}')
  42 ax2.set_xlabel('轮廓系数', fontsize=11); ax2.set_ylabel('样本索引', fontsize=11)
  43 ax2.legend(fontsize=8, loc='best')
  44 ax2.spines['top'].set_visible(False); ax2.spines['right'].set_visible(False)
  45 
  46 fig.tight_layout()
  47 save_fig(fig, 'figures/fig_kmeans_dual.pdf')
